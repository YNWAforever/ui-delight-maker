const facts = async () => {
  const v = {};
  for (const table of ["touchpoints", "agent_runs", "activity_logs", "agent_policy_versions"]) {
    const rows = (await pool.query('select row_to_json(t)::text value from "' + table + '" t')).rows
      .map((x) => x.value)
      .sort();
    v[table] = {
      count: rows.length,
      hash: createHash("sha256").update(JSON.stringify(rows)).digest("hex"),
    };
  }
  return v;
};

  assert.equal(new Set(identities.map((x) => x.userSha256)).size, 7);
  assert.equal(new Set(identities.map((x) => x.sessionSha256)).size, 7);
  assert.equal(new Set(identities.map((x) => x.cookiesSha256)).size, 7);
  assert.equal(pageErrors.length, 0);
  check("all seven identities/sessions/cookies distinct and zero uncaught browser errors");
  save(true);
  console.log(
    JSON.stringify({
      passed: true,
      checks: checks.length,
      roles: identities.length,
      sha: rt.sha,
      actualProviderCalls: 0,
    }),
  );
} catch (e) {
  save(false, { name: e.name, message: String(e.message).slice(0, 500) });
  console.log(
    JSON.stringify({
      passed: false,
      stage,
      checks: checks.length,
      error: String(e.message).slice(0, 500),
    }),
  );
  process.exitCode = 1;
} finally {
  await pool.end();
  await browser.close();
}
