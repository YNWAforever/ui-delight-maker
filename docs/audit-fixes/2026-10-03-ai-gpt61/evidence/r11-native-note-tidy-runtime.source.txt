const http = createServer(async (req, res) => {
  try {
    const u = new URL(req.url, "http://" + req.headers.host);
    assert.ok(["localhost", "127.0.0.1"].includes(u.hostname));
    if (req.headers.origin && new URL(req.headers.origin).origin !== u.origin) {
      res.writeHead(403);
      res.end("Foreign origin blocked");
      return;
    }
    if (!["GET", "HEAD", "POST"].includes(req.method)) {
      res.writeHead(405);
      res.end();
      return;
    }
    if (
      u.pathname.startsWith("/api/") &&
      u.pathname !== "/api/build" &&
      u.pathname !== "/api/auth/get-session"
    ) {
      res.writeHead(403);
      res.end("Non-UAT API blocked");
      return;
    }
    const file = resolve(assets, "." + decodeURIComponent(u.pathname));
    if (file.startsWith(assets + sep) && existsSync(file) && statSync(file).isFile()) {
      res.writeHead(200, { "content-type": mime[extname(file)] ?? "application/octet-stream" });
      res.end(req.method === "HEAD" ? undefined : readFileSync(file));
      return;
    }
    let size = 0;
    const chunks = [];
    for await (const chunk of req) {
      size += chunk.length;
      assert.ok(size < 1024 * 1024);
      chunks.push(chunk);
    }
    const headers = new Headers();
    for (const [k, v] of Object.entries(req.headers))
      if (v !== undefined) headers.set(k, Array.isArray(v) ? v.join(",") : v);
    const response = await app.fetch(
      new Request(u, {
        method: req.method,
        headers,
        ...(chunks.length ? { body: Buffer.concat(chunks) } : {}),
      }),
    );
    requests.push({
      at: new Date().toISOString(),
      path: u.pathname,
      method: req.method,
      status: response.status,
    });
    const out = Object.fromEntries(response.headers);
    res.writeHead(response.status, out);
    res.end(req.method === "HEAD" ? undefined : Buffer.from(await response.arrayBuffer()));
